cmake_minimum_required(VERSION 3.20)
project(VisionImageCodecs VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(ZLIB_USE_STATIC_LIBS ON)
find_package(ZLIB REQUIRED)
find_package(TIFF REQUIRED)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(WebP CONFIG REQUIRED)
add_library(VisionImageCodecs SHARED reader.c tiff_reader.c png_reader.c jpeg_reader.c webp_reader.c)
target_link_libraries(VisionImageCodecs PRIVATE TIFF::TIFF PNG::PNG JPEG::JPEG ZLIB::ZLIB WebP::webpdecoder)
target_compile_definitions(VisionImageCodecs PRIVATE VI_EXPORTS _FILE_OFFSET_BITS=64 _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(VisionImageCodecs PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(VisionImageCodecs PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
endif()
install(TARGETS VisionImageCodecs RUNTIME DESTINATION . LIBRARY DESTINATION .)
