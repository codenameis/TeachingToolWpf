회전은 두 성분을 섞는 계산

“점을 중심 주위로 돌린다는 말을 덧셈과 곱셈으로 어떻게 바꿀까요?” 시계의 바늘 끝을 생각하세요. 바늘의 길이는 유지하고 가로·세로로 뻗은 몫을 바꿉니다. 먼저 중심을 원점으로 옮긴 뒤 바늘을 돌리고, 중심을 원래 자리로 되돌립니다.

각도는 C# UI에서 도(degree), Math.Sin/Math.Cos에서 라디안(radian)을 씁니다. 한 바퀴 360°가 2π rad이므로 rad=degree×π180rad = degree × π/180입니다. cos는 가로 방향 몫, sin은 세로 방향 몫으로 생각하세요. 0°에서 cos=1,sin=0,90°cos=1, sin=0, 90°에서 cos≈0,sin=1cos≈0, sin=1입니다.

ROI 내부와 AABB는 다른 질문

“회전한 부품을 감싸는 상자 안을 클릭했는데 부품 바깥일 수도 있나요?” 기울어진 액자를 똑바른 택배 상자에 넣으면 모서리에 빈 공간이 남습니다. AABB는 이 택배 상자입니다. 빠르게 후보를 찾는 데 쓰고, 실제 포함 여부는 액자 모양으로 다시 판단합니다.

ROI(Region of Interest)는 관심 영역입니다. 회전 사각형 안에 마우스가 있는지 보려면 마우스를 ROI의 로컬 좌표로 역회전합니다. 그 뒤 |x|≤W2|x| ≤ W/2와 |y|≤H2|y| ≤ H/2를 검사합니다. 원은 중심과의 거리 제곱이 r2r² 이하인지, 폴리곤은 실제 변과 내부를 검사합니다.

namespace VisionGeometry;

/// <summary>기하·입력·렌더가 공유하는 이미지 좌표의 강체 변환(회전과 이동). 기본값은 항등 변환이다.</summary>
public readonly record struct RigidTransform2D
{
    public PointD Pivot
    {
        get;
    }

    public double AngleDegrees
    {
        get;
    }

    public PointD Translation
    {
        get;
    }

    private readonly double _cosine;

    private readonly double _sine;

    public double Cosine => AngleDegrees == 0 ? 1 : _cosine;

    public double Sine => _sine;

    public bool IsIdentity => AngleDegrees % 360 == 0 && Translation == default;

    public static RigidTransform2D Identity => default;

    public RigidTransform2D(PointD pivot, double angleDegrees, PointD translation = default)
    {
        if (!pivot.IsFinite || !translation.IsFinite || !double.IsFinite(angleDegrees))
        {
            throw new ArgumentOutOfRangeException(nameof(angleDegrees), "변환에는 유한한 좌표와 각도가 필요합니다.");
        }

        Pivot = pivot;
        Translation = translation;
        AngleDegrees = angleDegrees;
        var reduced = Math.IEEERemainder(angleDegrees, 360);
        (_sine, _cosine) = reduced == 0 ? (0, 1) : Math.SinCos(reduced * Math.PI / 180);
    }

    // 회전 중심을 원점으로 옮겨 회전한 다음 중심과 이동량을 더한다. p' = c + R(p - c) + t.
    public PointD Apply(PointD point) => AngleDegrees % 360 == 0
        ? point + Translation
        : Pivot + ApplyVector(point - Pivot) + Translation;

    public PointD ApplyVector(PointD vector) => new(
        (vector.X * Cosine) - (vector.Y * Sine),
        (vector.X * Sine) + (vector.Y * Cosine));

    public PointD InverseApply(PointD point)
    {
        if (AngleDegrees % 360 == 0)
        {
            return point - Translation;
        }

        // 역변환은 이동을 먼저 빼고 반대 각도로 회전한다. 원래 변환의 역순이어야 한다.
        var relative = point - Translation - Pivot;
        return Pivot + new PointD(
            (relative.X * Cosine) + (relative.Y * Sine),
            (-relative.X * Sine) + (relative.Y * Cosine));
    }

    public BoundsD TransformBounds(BoundsD bounds) => Enclose(Apply(new(bounds.Left, bounds.Top)), Apply(new(bounds.Right, bounds.Top)),
        Apply(new(bounds.Right, bounds.Bottom)), Apply(new(bounds.Left, bounds.Bottom)));

    public BoundsD InverseTransformBounds(BoundsD bounds) => Enclose(InverseApply(new(bounds.Left, bounds.Top)), InverseApply(new(bounds.Right, bounds.Top)),
        InverseApply(new(bounds.Right, bounds.Bottom)), InverseApply(new(bounds.Left, bounds.Bottom)));

    private static BoundsD Enclose(PointD a, PointD b, PointD c, PointD d) => new(
        Math.Min(Math.Min(a.X, b.X), Math.Min(c.X, d.X)), Math.Min(Math.Min(a.Y, b.Y), Math.Min(c.Y, d.Y)),
        Math.Max(Math.Max(a.X, b.X), Math.Max(c.X, d.X)), Math.Max(Math.Max(a.Y, b.Y), Math.Max(c.Y, d.Y)));
}

using System.Collections.Immutable;
namespace VisionGeometry;

public sealed record RoiGeometry
{
    public PointD Center
    {
        get;
    }

    public double Width
    {
        get;
    }

    public double Height
    {
        get;
    }

    public double Angle
    {
        get;
    }

    public RoiShape Shape
    {
        get;
    }

    public double Area => Width * Height * Shape.AreaFactor;

    public double Radius => Shape.Radius(Width);

    public PointD ContainmentAnchor => Radius > 0 ? Center : ToWorld(new(Shape.Vertices[0].X * Width, Shape.Vertices[0].Y * Height));

    public PointD[] Outline => Shape.Vertices.Select(p => ToWorld(new(p.X * Width, p.Y * Height))).ToArray();

    public string ShapeName => Shape.Kind switch
    {
        RoiShapeKind.Circle => "원",
        RoiShapeKind.Triangle => "삼각형",
        RoiShapeKind.Polygon => "폴리곤",
        _ => "사각형"
    };

    public RoiGeometry(PointD center, double width, double height, double angle, RoiShape? shape = null)
    {
        if (!center.IsFinite)
        {
            throw new ArgumentOutOfRangeException(nameof(center));
        }

        if (!double.IsFinite(width) || width < 0.01)
        {
            throw new ArgumentOutOfRangeException(nameof(width));
        }

        if (!double.IsFinite(height) || height < 0.01)
        {
            throw new ArgumentOutOfRangeException(nameof(height));
        }

        if (!double.IsFinite(angle) || !double.IsFinite(width * height))
        {
            throw new ArgumentOutOfRangeException(nameof(angle));
        }

        Center = center;
        Width = width;
        Height = height;
        Angle = NormalizeAngle(angle);
        Shape = shape ?? RoiShape.Rectangle;
        if (!double.IsFinite(Area) || Area <= 0)
        {
            throw new ArgumentException("도형 면적은 유한한 양수여야 합니다.", nameof(shape));
        }

        if (Shape.Kind == RoiShapeKind.Circle && width != height)
        {
            throw new ArgumentException("원의 너비와 높이는 지름으로 같아야 합니다.");
        }
    }

    public RoiGeometry WithFrame(PointD center, double width, double height, double angle)
        => new(center, width, height, angle, Shape);

    /// <summary>기존 각도를 유지하며 모든 도형을 감싸는 사각형으로 확장한다. 이미 포함하면 원래 도형을 반환한다.</summary>
    public RoiGeometry ExpandToCover(IEnumerable<RoiGeometry> members)
    {
        var xAxis = new PointD(1, 0).Rotate(Angle);
        var yAxis = new PointD(0, 1).Rotate(Angle);
        var originX = Dot(Center, xAxis);
        var originY = Dot(Center, yAxis);
        var left = -Width / 2;
        var right = Width / 2;
        var top = -Height / 2;
        var bottom = Height / 2;
        var expanded = false;
        foreach (var member in members)
        {
            expanded |= !Covers(member);
            left = Math.Min(left, -member.Support(xAxis * -1) - originX);
            right = Math.Max(right, member.Support(xAxis) - originX);
            top = Math.Min(top, -member.Support(yAxis * -1) - originY);
            bottom = Math.Max(bottom, member.Support(yAxis) - originY);
        }
        var bounds = new BoundsD(left, top, right, bottom);
        return expanded ? new(ToWorld(bounds.Center), bounds.Width, bounds.Height, Angle) : this;
    }

    public static RoiGeometry Create(RoiShapeKind kind, PointD center, double width, double height)
    {
        if (!Enum.IsDefined(kind))
        {
            throw new ArgumentOutOfRangeException(nameof(kind));
        }

        if (kind == RoiShapeKind.Circle)
        {
            return new(center, Math.Min(width, height), Math.Min(width, height), 0, RoiShape.Circle);
        }

        if (kind == RoiShapeKind.Triangle)
        {
            return new(center, width, height, 0, RoiShape.Triangle);
        }

        if (kind == RoiShapeKind.Polygon)
        {
            return Polygon(Enumerable.Range(0, 5).Select(i => center + new PointD(Math.Sin(i * Math.PI * 2 / 5) * width / 2, -Math.Cos(i * Math.PI * 2 / 5) * height / 2)));
        }

        return new(center, width, height, 0);
    }

    public static RoiGeometry Polygon(IEnumerable<PointD> vertices)
    {
        var points = vertices.ToArray();
        PolygonMath.Validate(points);
        var bounds = BoundsD.FromPoints(points);
        var normalized = points.Select(p => new PointD((p.X - bounds.Center.X) / bounds.Width, (p.Y - bounds.Center.Y) / bounds.Height)).ToImmutableArray();
        return new(bounds.Center, bounds.Width, bounds.Height, 0, new RoiShape.PolygonShape(RoiShapeKind.Polygon, normalized));
    }

    public double Support(PointD direction) => (Center.X * direction.X) + (Center.Y * direction.Y)
        + Shape.Support(direction.Rotate(-Angle), Width, Height);

    public static double NormalizeAngle(double angle) => ((angle % 360) + 360) % 360;

    public PointD ToWorld(PointD local) => Center + local.Rotate(Angle);

    public PointD ToLocal(PointD world) => (world - Center).Rotate(-Angle);

    public PointD[] FrameCorners =>
    [
        ToWorld(new(-Width / 2, -Height / 2)), ToWorld(new(Width / 2, -Height / 2)),
        ToWorld(new(Width / 2, Height / 2)), ToWorld(new(-Width / 2, Height / 2))
    ];

    public BoundsD Bounds
    {
        get
        {
            if (Radius > 0)
            {
                return new(Center.X - Radius, Center.Y - Radius, Center.X + Radius, Center.Y + Radius);
            }

            if (Shape.Kind == RoiShapeKind.Rectangle)
            {
                // 회전 사각형을 감싸는 축 정렬 경계. 실제 ROI 내부 판정은 Contains가 담당한다.
                var (sine, cosine) = Math.SinCos(Angle * Math.PI / 180);
                var halfBoundsWidth = ((Width * Math.Abs(cosine)) + (Height * Math.Abs(sine))) / 2;
                var halfBoundsHeight = ((Width * Math.Abs(sine)) + (Height * Math.Abs(cosine))) / 2;
                return new(Center.X - halfBoundsWidth, Center.Y - halfBoundsHeight,
                    Center.X + halfBoundsWidth, Center.Y + halfBoundsHeight);
            }
            return new(-Support(new(-1, 0)), -Support(new(0, -1)), Support(new(1, 0)), Support(new(0, 1)));
        }
    }

    public bool Contains(PointD world, double tolerance = 0)
    {
        var local = ToLocal(world);
        return Shape.ContainsLocal(local, Width, Height, tolerance);
    }

    public bool Contains(RoiGeometry child)
    {
        var tolerance = Math.Max(1e-6, Math.Max(Math.Abs(Center.X), Math.Abs(Center.Y)) * 1e-12);
        if (Area <= child.Area + (tolerance * (Width + Height)))
        {
            return false;
        }

        return Covers(child);
    }

    /// <summary>Actual shape containment including equal boundaries; drawing uses the stricter Contains check.</summary>
    public bool Covers(RoiGeometry child)
    {
        var tolerance = Math.Max(1e-6, Math.Max(Math.Abs(Center.X), Math.Abs(Center.Y)) * 1e-12);
        if (Radius > 0)
        {
            return child.Radius > 0 ? (Center - child.Center).Length + child.Radius <= Radius + tolerance
                : child.Outline.All(p => Contains(p, tolerance));
        }

        if (Shape.Kind == RoiShapeKind.Rectangle)
        {
            var x = new PointD(1, 0).Rotate(Angle);
            var y = new PointD(0, 1).Rotate(Angle);
            return child.Support(x) - Dot(Center, x) <= (Width / 2) + tolerance &&
                child.Support(x * -1) + Dot(Center, x) <= (Width / 2) + tolerance &&
                child.Support(y) - Dot(Center, y) <= (Height / 2) + tolerance &&
                child.Support(y * -1) + Dot(Center, y) <= (Height / 2) + tolerance;
        }
        var boundary = Outline;
        if (child.Radius > 0)
        {
            return Contains(child.Center, tolerance) && Enumerable.Range(0, boundary.Length)
                .All(i => PolygonMath.Distance(child.Center, boundary[i], boundary[(i + 1) % boundary.Length]) + tolerance >= child.Radius);
        }

        var childBoundary = child.Outline;
        return childBoundary.All(p => Contains(p, tolerance)) && Enumerable.Range(0, childBoundary.Length)
            .All(i => PolygonMath.ContainsSegment(boundary, childBoundary[i], childBoundary[(i + 1) % childBoundary.Length], tolerance));
    }

    private static double Dot(PointD a, PointD b) => (a.X * b.X) + (a.Y * b.Y);

    public RoiGeometry Translate(PointD delta) => WithFrame(Center + delta, Width, Height, Angle);

    public RoiGeometry RotateAround(PointD pivot, double angle) =>
        WithFrame(pivot + (Center - pivot).Rotate(angle), Width, Height, Angle + angle);
}

절댓값과 반너비가 필요한 이유

W와 H는 ROI의 원래 너비·높이(px)입니다. 가로 변의 회전 후 가로축 성분은 W cos θ, 세로 변의 회전 후 가로축 성분은 −H sin θ입니다. 바깥 상자의 길이를 구할 때는 방향의 부호가 아니라 두 성분의 길이를 더하므로 절댓값을 씁니다. 전체 너비를 2로 나누면 중심에서 한쪽 끝까지의 반너비입니다.

namespace VisionGeometry;

/// <summary>원본 이미지 좌표와 뷰포트 DIP 좌표를 변환한다. 물리 픽셀/DPI 변환은 호스트가 담당한다.</summary>
public readonly record struct ViewportTransform(double Scale, PointD Offset)
{
    public static ViewportTransform Identity => new(1, new(0, 0));

    public PointD ImageToScreen(PointD point) => (point * Scale) + Offset;

    public PointD ScreenToImage(PointD point) => (point - Offset) / Scale;

    public ViewportTransform Pan(PointD delta) => this with { Offset = Offset + delta };

    public ViewportTransform ZoomAt(PointD anchor, double factor)
    {
        if (!double.IsFinite(factor) || factor <= 0)
        {
            throw new ArgumentOutOfRangeException(nameof(factor));
        }

        var scale = Math.Clamp(Scale * factor, 0.00001, 128);
        // anchor 아래의 원본 좌표가 확대 후에도 같은 화면 위치에 오도록 Offset을 역산한다.
        return new(scale, anchor - (ScreenToImage(anchor) * scale));
    }

    public static ViewportTransform Fit(double imageWidth, double imageHeight, double viewWidth, double viewHeight)
    {
        if (imageWidth <= 0 || imageHeight <= 0 || viewWidth <= 0 || viewHeight <= 0)
        {
            return Identity;
        }

        // 두 축 모두 들어가는 배율에 여백을 두고, 남은 공간의 절반만큼 이동해 중앙에 둔다.
        var scale = Math.Min(viewWidth / imageWidth, viewHeight / imageHeight) * 0.92;
        return new(scale, new(
            (viewWidth - (imageWidth * scale)) / 2,
            (viewHeight - (imageHeight * scale)) / 2));
    }
}
