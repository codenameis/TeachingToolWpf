5. 이제 실제 타일 계획 코드를 읽기

수식과 이름을 연결하세요. downsampleFactor는 2L2의 L제곱, tileSpanInImage는 S, firstColumn/columnEnd는 시작/제외 끝입니다. 1 << level은 정수 1을 왼쪽으로 L비트 옮겨 2L을 만드는 표현입니다. 이 메서드는 주소 목록만 계산하며 파일 읽기나 GPU 생성은 하지 않습니다.

TilePlanner.cs · 41줄 · 전체 파일
using VisionGeometry;
using VisionImaging.Abstractions;

namespace VisionCanvas.Imaging;

public static class TilePlanner
{
    /// <summary>뷰포트에 필요한 타일 주소를 행 우선 순서로 구한다. 픽셀 읽기와 캐시 관리는 하지 않는다.</summary>
    public static IReadOnlyList<TileKey> Visible(ImageInfo image, ViewportTransform view, double width, double height, double dpi)
    {
        if (width <= 0 || height <= 0 || view.Scale <= 0 || dpi <= 0)
        {
            return [];
        }

        // Scale은 DIP/원본 픽셀, dpi는 물리 픽셀/DIP다. 곱한 값으로 표시 해상도를 고른다.
        var level = Math.Clamp((int)Math.Floor(Math.Log2(1 / (view.Scale * dpi))), 0, image.Levels - 1);
        var downsampleFactor = 1 << level;
        var tileSpanInImage = (double)ImageInfo.TileSize * downsampleFactor;
        var imageTopLeft = view.ScreenToImage(new PointD(0, 0));
        var imageBottomRight = view.ScreenToImage(new PointD(width, height));
        var columnCount = (int)Math.Ceiling(image.Width / tileSpanInImage);
        var rowCount = (int)Math.Ceiling(image.Height / tileSpanInImage);

        // 시작은 포함하고 끝은 제외한다. 이미지 밖 여백에는 타일을 요청하지 않는다.
        var firstColumn = Math.Clamp((int)Math.Floor(imageTopLeft.X / tileSpanInImage), 0, columnCount);
        var firstRow = Math.Clamp((int)Math.Floor(imageTopLeft.Y / tileSpanInImage), 0, rowCount);
        var columnEnd = Math.Clamp((int)Math.Ceiling(imageBottomRight.X / tileSpanInImage), 0, columnCount);
        var rowEnd = Math.Clamp((int)Math.Ceiling(imageBottomRight.Y / tileSpanInImage), 0, rowCount);
        var result = new List<TileKey>();
        for (var row = firstRow; row < rowEnd; row++)
        {
            for (var column = firstColumn; column < columnEnd; column++)
            {
                result.Add(new(level, column, row));
            }
        }

        return result;
    }
}

  2. 장면에서 주소를 구한다. GpuImageResources는 가장 작은 전체 개요 타일 (Levels−1,0,0)을 먼저 붙이고, Planner가 구한 상세 타일 목록과 중복을 제거합니다.
  3. 캐시에 없는 주소만 요청한다. 이미 GPU에 있는 타일은 재사용합니다. 현재 목록은 개요 다음에 행/열 순이며, 커서 중심 우선이나 이동 방향 예측을 구현한 것으로 설명하지 않습니다.
  4. TileWorker 한 곳에서 읽는다. 대기 요청은 최신 목록으로 합칩니다. 새 요청이 생겼다고 실행 중인 Read를 즉시 중단하는 것은 아니며, 현재 타일 처리 뒤 최신 요청으로 넘어갑니다. 준비 슬롯 32개가 가득 차면 소비를 기다립니다.
  5. 렌더 측에서 세대를 확인한다. 한 번의 Update에서 준비 결과를 최대16개 꺼내고 현재 세대·중복을 검사해 업로드합니다. 꺼낸 CPU 버퍼는 이 범위가 끝날 때 반환합니다.
  6. 완료된 조각을 그린다. 개요가 준비되어 있으면 상세 타일이 아직 없는 곳의 배경으로 활용합니다. 준비된 불투명 상세 타일이 모두 덮는 경우 개요 중복 그리기를 생략합니다. 개요마저 아직 없다면 즉시 완성 영상을 보장하지 않습니다.

/// <summary>One decoding owner, a coalesced request and at most 32 prepared tiles.</summary>
public sealed class TileWorker : IDisposable
{
    private readonly object _gate = new();
    private readonly AutoResetEvent _wake = new(false);
    private readonly SemaphoreSlim _slots = new(32, 32);
    private readonly CancellationTokenSource _shutdown = new();
    private readonly ConcurrentQueue<ReadyTile> _ready = new();
    private readonly Thread _thread;
    private ITileSource? _pendingSource;
    private long _generation;
    private Request? _pendingRequest;
    private sealed record Request(long Generation, IReadOnlyList<TileKey> Keys);
    public event Action? TileReady;
    public event Action<string>? Failed;

    public long Generation => Interlocked.Read(ref _generation);

    public bool HasReady => !_ready.IsEmpty;

    public TileWorker()
    {
        _thread = new Thread(Run) { IsBackground = true, Name = "Image tile decoder" };
        _thread.Start();
    }

    public long SetSource(ITileSource source)
    {
        lock (_gate)
        {
            _pendingSource?.Dispose();
            _pendingSource = source;
            _pendingRequest = null;
            Interlocked.Increment(ref _generation);
        }
        _wake.Set();
        return Generation;
    }

    public void RequestTiles(long expectedGeneration, IReadOnlyList<TileKey> keys)
    {
        if (keys.Count == 0)
        {
            return;
        }

        lock (_gate)
        {
            _pendingRequest = new(expectedGeneration, keys);
        }

        _wake.Set();
    }

    public bool TryTake(out ReadyTile? tile)
    {
        if (!_ready.TryDequeue(out tile))
        {
            return false;
        }

        _slots.Release();
        return true;
    }

    private void Run()
    {
        ITileSource? source = null;
        long sourceGeneration = 0;
        long failedGeneration = -1;
        try
        {
            while (!_shutdown.IsCancellationRequested)
            {
                _wake.WaitOne();
                Request? request;
                lock (_gate)
                {
                    if (_pendingSource is { } next)
                    {
                        source?.Dispose();
                        source = next;
                        _pendingSource = null;
                        sourceGeneration = _generation;
                    }
                    request = _pendingRequest;
                    _pendingRequest = null;
                }
                if (source is null || request is null || request.Generation != sourceGeneration || sourceGeneration == failedGeneration)
                {
                    continue;
                }

                foreach (var key in request.Keys)
                {
                    if (Generation != sourceGeneration || _shutdown.IsCancellationRequested)
                    {
                        break;
                    }

                    _slots.Wait(_shutdown.Token);
                    PixelBuffer? pixels = null;
                    var queued = false;
                    try
                    {
                        pixels = source.Read(key, _shutdown.Token);
                        if (Generation == sourceGeneration)
                        {
                            _ready.Enqueue(new(sourceGeneration, key, pixels));
                            pixels = null;
                            queued = true;
                        }
                    }
                    catch (OperationCanceledException) when (_shutdown.IsCancellationRequested)
                    {
                        throw;
                    }
                    catch (Exception ex)
                    {
                        failedGeneration = sourceGeneration;
                        Failed?.Invoke($"타일 읽기 실패: {ex.Message}. 이미지를 다시 열어 재시도하세요.");
                        break;
                    }
                    finally
                    {
                        pixels?.Dispose();
                        // A queued buffer owns the slot until TryTake. Notification is not part of decoding.
                        if (!queued)
                        {
                            _slots.Release();
                        }
                    }
                    if (queued)
                    {
                        TileReady?.Invoke();
                    }

                    lock (_gate)
                    {
                        if (_pendingRequest is not null)
                        {
                            break;
                        }
                    }
                }
            }
        }
        catch (OperationCanceledException) when (_shutdown.IsCancellationRequested)
        {
        }
        catch (Exception ex)
        {
            Failed?.Invoke($"타일 읽기 실패: {ex.Message}");
        }
        finally
        {
            source?.Dispose();
        }
    }

    public void Dispose()
    {
        _shutdown.Cancel();
        _wake.Set();
        _thread.Join();
        lock (_gate)
        {
            _pendingSource?.Dispose();
            _pendingSource = null;
        }
        while (_ready.TryDequeue(out var item))
        {
            item.Pixels.Dispose();
        }

        _shutdown.Dispose();
        _wake.Dispose();
        _slots.Dispose();
    }
}

“같은 배열을 두 곳이 받았으니 둘 다 Dispose하면 더 안전할까요?” 도서관 책을 한 권 빌려 두 사람이 같이 읽어도 반납은 한 번이어야 합니다. 배열 풀에서도 누군가 반납하면 다른 사용자가 그 배열을 빌려 덮어쓸 수 있습니다. 참조를 가지고 있다는 사실만으로 계속 읽어도 된다는 뜻은 아닙니다.

/// <summary>Single-slot ownership transfer. Replaced frames are returned immediately.</summary>
public sealed class LatestFrameMailbox : IDisposable
{
    private readonly object _gate = new();
    private VideoFrame? _pending;
    private VideoFrame? _latest;
    private bool _disposed;
    private long _dropped;

    public long Dropped => Interlocked.Read(ref _dropped);

    public bool HasPending
    {
        get
        {
            lock (_gate)
            {
                return _pending is not null;
            }
        }
    }

    public event Action? Available;

    public void Publish(VideoFrame frame)
    {
        lock (_gate)
        {
            if (_disposed)
            {
                frame.Dispose();
                return;
            }
            if (_pending is not null)
            {
                _pending.Dispose();
                Interlocked.Increment(ref _dropped);
            }
            _latest?.Dispose();
            _latest = frame.Retain();
            _pending = frame;
            Available?.Invoke();
        }
    }

    public VideoFrame? Take()
    {
        lock (_gate)
        {
            var frame = _pending;
            _pending = null;
            return frame;
        }
    }

    public VideoFrame? RetainLatest()
    {
        lock (_gate)
        {
            return _latest?.Retain();
        }
    }

    public void Clear()
    {
        lock (_gate)
        {
            _pending?.Dispose();
            _latest?.Dispose();
            _pending = null;
            _latest = null;
        }
    }

    public void Dispose()
    {
        lock (_gate)
        {
            _disposed = true;
            Clear();
            Available = null;
        }
    }
}

        {
            get;
        } = pixels;

        public void Retain()
        {
            int count;
            do
            {
                count = Volatile.Read(ref _references);
                if (count == 0)
                {
                    throw new ObjectDisposedException(nameof(VideoFrame));
                }
            }
            while (Interlocked.CompareExchange(ref _references, count + 1, count) != count);
        }

        public void Release()
        {
            if (Interlocked.Decrement(ref _references) == 0)
            {
                Pixels.Dispose();
            }
        }
    }

    private int _disposed;

        get;
    }

    {
        get;
    }

표시는 최신 완료 프레임 우선

생산자가 60장/초, 표시가 30장/초라면 모든 프레임을 줄 세워 표시할 때 지연이 계속 커집니다. 한 칸 mailbox는 아직 소비되지 않은 프레임을 최신 것으로 바꾸고 이전 lease를 반환합니다. 버린 개수는 Dropped로 셉니다. ‘모든 영상을 저장·검사해야 한다’는 무손실 요구는 다른 파이프라인입니다.

기본 업로드는 CPU 원본 → 재사용 upload buffer → GPU texture입니다. Retain이 전체 픽셀 복사를 피한다고 전체 파이프라인이 zero-copy가 되지는 않습니다. 풀 배열·upload 공간·GPU texture·타일 캐시·디스크 캐시의 메모리는 각각 계산해야 합니다.

직접 확인: A 타일 준비 중 B를 열었다. A 타일이 완료되면?

세대가 다르면 B 장면에 넣지 않고 A의 PixelBuffer를 반환합니다. B의 source는 유지합니다. 취소가 늦게 반영되어 A의 계산이 끝나도 게시 검증이 막아야 합니다. 예외를 무시하는 것으로 이 계약을 대신하지 않습니다.

7. 캐시와 메모리 예산을 함께 설계하기

캐시는 다시 볼 가능성이 있는 타일을 보관하는 공간입니다. LRU는 Least Recently Used, 즉 가장 오래 사용하지 않은 항목부터 버리는 방식입니다. 현재 구현은 마지막 사용 프레임 번호를 기록하고 현재 프레임에서 사용하는 타일은 퇴출하지 않습니다. 같은 번호의 항목 사이 순서까지 특별한 우선순위라고 해석하지 않습니다.

        if (scene.IsLive && frame is not null)
        {
            var key = (scene.Document.Viewport, scene.Width, scene.Height, scene.Dpi);
            if (ReferenceEquals(frame, _preparedIdentity) && key == _preparedView)
            {
                return;
            }

            using var prepared = preparer.Prepare(frame.Pixels, key.Viewport, key.Width, key.Height, key.Dpi);
            _preparedIdentity = frame;
            _preparedView = key;
            if (prepared is null)
            {
                Live = null;
                return;
            }
            var index = surface.FrameIndex;
            var pixels = prepared.Pixels;
            var texture = _video[index];
            if (texture is null || texture.Width != pixels.Width || texture.Height != pixels.Height || texture.Kind != pixels.Kind)
            {
                if (texture is not null)
                {
                    surface.Retire(texture);
                }

                _video[index] = texture = pipeline.CreateTexture(pixels.Width, pixels.Height, pixels.Kind);
            }
            pipeline.Update(texture, pixels, surface);
            Live = texture;
            LiveBounds = prepared.ImageBounds;
        }
        else if (!scene.IsLive && _preparedIdentity is not null)
        {
            for (var index = 0; index < 2; index++)
            {
                if (_video[index] is { } texture)
                {
                    surface.Retire(texture);
                }

                _video[index] = null;
            }
            Live = null;
            _preparedIdentity = null;
        }
    }

    private void UpdateTiles(ImageInfo image, RenderScene scene, Dx12Surface surface, long frameNumber)
    {
        var visible = TilePlanner.Visible(image, scene.Document.Viewport, scene.Width, scene.Height, scene.Dpi).Prepend(new TileKey(image.Levels - 1, 0, 0)).Distinct().ToArray();
        foreach (var key in visible)
        {
            if (_tiles.ContainsKey(key))
            {
                _lastUsed[key] = frameNumber;
            }
        }

        var bytes = _tiles.Values.Sum(t => t.Bytes);
        if (bytes > gpu.TileBudgetBytes)
        {
            foreach (var entry in _lastUsed.OrderBy(p => p.Value).ToArray())
            {
                if (bytes <= gpu.TileBudgetBytes)
                {
                    break;
                }

                if (entry.Value == frameNumber)
                {
                    continue;
                }

                var texture = _tiles[entry.Key];
                bytes -= texture.Bytes;
                _tiles.Remove(entry.Key);
                _lastUsed.Remove(entry.Key);
                surface.Retire(texture);
            }
        }
        source.RequestTiles(_generation, visible.Where(key => !_tiles.ContainsKey(key)).ToArray());
    }

        foreach (var texture in _video)

        }
    }

8. 타일 경계의 줄과 필터 이음새 구별하기

“각 타일에서 평균 필터를 돌렸더니 조각 사이에 선이 생겨요.” 조각 경계의 픽셀도 옆 조각의 이웃이 필요합니다. 책을 두 권으로 나눴다고 문장 중간에서 앞뒤 단어를 잃어서는 안 되는 것과 같습니다. 계산할 영역보다 조금 넓게 읽는 여백을 halo라고 부릅니다.

예를 들어 3×33×3 평균 필터는 좌우·상하 한 픽셀 이웃이 필요합니다.512×512. 512×512 출력 타일을 독립 처리하려면 주변까지 514×514514×514를 읽어 계산한 뒤 가운데 512×512512×512만 남길 수 있습니다. 이 추가 테두리를 halo라고 부릅니다. 반경 r이면 내부 타일의 읽기 폭은 T+2r이며, 실제 영상 끝에서만 반사·복제·상수 같은 경계 규칙을 적용합니다. 매 타일 끝을 영상 끝처럼 처리하면 이웃 타일의 실제 픽셀 대신 가짜 경계를 사용합니다.

가로 한 줄의 평균 예로 타일 A의 마지막 값이 0, 타일 B의 앞 두 값이 100,100이라면 B 첫 위치의 3점 평균은 (0+100+100)3≈66.7(0+100+100)/3≈66.7입니다. B의 가장자리 값을 복제하면 (100+100+100)3=100(100+100+100)/3=100이 되어 경계에 차이가 남습니다. 여러 필터를 연결할 때는 각 연산의 이웃 범위도 누적해서 고려해야 합니다.

이 halo 사례는 직접 타일 필터를 설계할 때의 교육 예이며, 현재 Canvas의 타일 표시 경로에 추가된 기능이 아닙니다. 영상 바깥 padding 규칙은 OpenCV 경계 처리 공식 설명을 참고하세요.

작게 표시한 타일을 검사 원본으로 쓰지 않습니다. 표시 피라미드는 보기 위한 데이터이고, 검사 학습·최종 검증은 별도 원본 reader와 유효 마스크 계약을 따릅니다.
직접 확인: 1300×9001300×900의 마지막 타일을 만들 때 512×512512×512를 채우면 왜 문제가 될까?

실제 유효 영역은 276×388276×388입니다. 나머지를 실제 검은 픽셀로 평균·검사하면 값과 도형 경계가 달라집니다. 고정 크기 저장소를 설계할 수는 있지만 유효 너비/높이를 별도로 전달하고 모든 소비자가 지켜야 합니다. 현재 PixelBuffer는 유효 크기로 생성하므로 이런 별도 padding을 영상으로 전달하지 않습니다.

가로 세 값 [0,100,100]의 평균은 약 66.7입니다. 왼쪽 0이 옆 타일에 있다는 이유로 현재 타일의 100을 복제하면 [100,100,100]이 되어 100이 나옵니다. 조각 경계마다 밝기가 바뀌는 원인입니다. 실제 영상 바깥에서만 정한 padding 규칙을 적용해야 합니다. 이는 필터 설계 설명이며 현재 Canvas에 새로운 타일 필터 기능을 추가했다는 뜻은 아닙니다.
