using System;

namespace MiniVisionCanvas;

public readonly record struct PointD(double X, double Y)
{
    public static PointD operator +(PointD a, PointD b) => new(a.X + b.X, a.Y + b.Y);
    public static PointD operator -(PointD a, PointD b) => new(a.X - b.X, a.Y - b.Y);
    public static PointD operator *(PointD a, double s) => new(a.X * s, a.Y * s);
    public static PointD operator /(PointD a, double s) => new(a.X / s, a.Y / s);

    public double LengthSquared => (X * X) + (Y * Y);
    public double Length => Math.Sqrt(LengthSquared);

    public static double Dot(PointD a, PointD b) => (a.X * b.X) + (a.Y * b.Y);
    public static double Cross(PointD a, PointD b) => (a.X * b.Y) - (a.Y * b.X);

    public PointD Normalize()
    {
        double len = Length;
        return len > 1e-12 ? new PointD(X / len, Y / len) : new PointD(1.0, 0.0);
    }
}

public readonly record struct BoundsD(double Left, double Top, double Right, double Bottom)
{
    public double Width => Math.Max(0.0, Right - Left);
    public double Height => Math.Max(0.0, Bottom - Top);
    public PointD Center => new((Left + Right) * 0.5, (Top + Bottom) * 0.5);

    public static BoundsD FromCenterSize(PointD center, double width, double height) =>
        new(center.X - (width * 0.5), center.Y - (height * 0.5),
            center.X + (width * 0.5), center.Y + (height * 0.5));

    public bool Contains(PointD pt) =>
        pt.X >= Left && pt.X <= Right && pt.Y >= Top && pt.Y <= Bottom;

    public bool Contains(BoundsD other) =>
        other.Left >= Left && other.Right <= Right && other.Top >= Top && other.Bottom <= Bottom;

    public bool Intersects(BoundsD other) =>
        Left <= other.Right && Right >= other.Left && Top <= other.Bottom && Bottom >= other.Top;

    public BoundsD Union(BoundsD other) =>
        new(Math.Min(Left, other.Left), Math.Min(Top, other.Top),
            Math.Max(Right, other.Right), Math.Max(Bottom, other.Bottom));

    public BoundsD Inflate(double margin) =>
        new(Left - margin, Top - margin, Right + margin, Bottom + margin);
}

public readonly record struct RigidTransform2D(double Cos, double Sin, double Tx, double Ty)
{
    public static RigidTransform2D Identity { get; } = new(1.0, 0.0, 0.0, 0.0);

    public static RigidTransform2D FromAngle(double radians, PointD translation) =>
        new(Math.Cos(radians), Math.Sin(radians), translation.X, translation.Y);

    public PointD Apply(PointD local) =>
        new((local.X * Cos) - (local.Y * Sin) + Tx,
            (local.X * Sin) + (local.Y * Cos) + Ty);

    public PointD ApplyInverse(PointD world)
    {
        double dx = world.X - Tx;
        double dy = world.Y - Ty;
        return new PointD((dx * Cos) + (dy * Sin), (-dx * Sin) + (dy * Cos));
    }

    public static PointD RotateAround(PointD pt, PointD pivot, double deltaRadians)
    {
        double cos = Math.Cos(deltaRadians);
        double sin = Math.Sin(deltaRadians);
        double dx = pt.X - pivot.X;
        double dy = pt.Y - pivot.Y;
        return new PointD(pivot.X + (dx * cos) - (dy * sin), pivot.Y + (dx * sin) + (dy * cos));
    }
}

public readonly record struct ViewportTransform(double Scale, double OffsetX, double OffsetY)
{
    public static ViewportTransform Identity { get; } = new(1.0, 20.0, 20.0);

    public PointD ImageToScreen(PointD img) =>
        new((img.X * Scale) + OffsetX, (img.Y * Scale) + OffsetY);

    public PointD ScreenToImage(PointD scr) =>
        new((scr.X - OffsetX) / Scale, (scr.Y - OffsetY) / Scale);

    public BoundsD ScreenToImage(BoundsD scrBounds)
    {
        PointD tl = ScreenToImage(new PointD(scrBounds.Left, scrBounds.Top));
        PointD br = ScreenToImage(new PointD(scrBounds.Right, scrBounds.Bottom));
        return new BoundsD(tl.X, tl.Y, br.X, br.Y);
    }

    public ViewportTransform Pan(double dxDip, double dyDip) =>
        new(Scale, OffsetX + dxDip, OffsetY + dyDip);

    public ViewportTransform ZoomAt(PointD cursorScreenDip, double factor)
    {
        double nextScale = Math.Clamp(Scale * factor, 0.02, 64.0);
        PointD anchorImg = ScreenToImage(cursorScreenDip);
        double nextOffsetX = cursorScreenDip.X - (anchorImg.X * nextScale);
        double nextOffsetY = cursorScreenDip.Y - (anchorImg.Y * nextScale);
        return new ViewportTransform(nextScale, nextOffsetX, nextOffsetY);
    }

    public static ViewportTransform Fit(double imgW, double imgH, double viewW, double viewH, double pad = 24.0)
    {
        if (imgW <= 0 || imgH <= 0 || viewW <= 0 || viewH <= 0) return Identity;
        double availW = Math.Max(1.0, viewW - (pad * 2.0));
        double availH = Math.Max(1.0, viewH - (pad * 2.0));
        double scale = Math.Clamp(Math.Min(availW / imgW, availH / imgH), 0.02, 64.0);
        double ox = (viewW - (imgW * scale)) * 0.5;
        double oy = (viewH - (imgH * scale)) * 0.5;
        return new ViewportTransform(scale, ox, oy);
    }

    public static ViewportTransform FocusBounds(BoundsD target, double viewW, double viewH, double pad = 40.0)
    {
        double w = Math.Max(16.0, target.Width);
        double h = Math.Max(16.0, target.Height);
        double availW = Math.Max(1.0, viewW - (pad * 2.0));
        double availH = Math.Max(1.0, viewH - (pad * 2.0));
        double scale = Math.Clamp(Math.Min(availW / w, availH / h), 0.02, 64.0);
        PointD c = target.Center;
        return new ViewportTransform(scale, (viewW * 0.5) - (c.X * scale), (viewH * 0.5) - (c.Y * scale));
    }
}

Step 2 — Step2_RoiGeometryAndPolygon.cs RoiGeometry (ToWorld / Contains / Support) & PolygonClosure 
using System;
using System.Collections.Generic;
using System.Collections.Immutable;

namespace MiniVisionCanvas;

public enum RoiShapeKind
{
    Rectangle,
    Circle,
    Polygon
}

public sealed record RoiGeometry(
    RoiShapeKind Shape,
    PointD Center,
    double Width,
    double Height,
    double RotationRadians,
    ImmutableArray<PointD> NormalizedVertices)
{
    public static RoiGeometry CreateRectangle(PointD center, double width, double height, double rotationRad = 0.0) =>
        new(RoiShapeKind.Rectangle, center, Math.Max(4.0, width), Math.Max(4.0, height), rotationRad, ImmutableArray<PointD>.Empty);

    public static RoiGeometry CreateCircle(PointD center, double width, double height, double rotationRad = 0.0) =>
        new(RoiShapeKind.Circle, center, Math.Max(4.0, width), Math.Max(4.0, height), rotationRad, ImmutableArray<PointD>.Empty);

    public PointD ToWorld(double u, double v)
    {
        RigidTransform2D pose = RigidTransform2D.FromAngle(RotationRadians, Center);
        return pose.Apply(new PointD(u * Width, v * Height));
    }

    public PointD ToLocalNormalized(PointD worldPt)
    {
        RigidTransform2D pose = RigidTransform2D.FromAngle(RotationRadians, Center);
        PointD unrotated = pose.ApplyInverse(worldPt);
        return new PointD(
            Width > 1e-9 ? unrotated.X / Width : 0.0,
            Height > 1e-9 ? unrotated.Y / Height : 0.0);
    }

    public bool Contains(PointD worldPt, double extraMarginPx = 0.0)
    {
        RigidTransform2D pose = RigidTransform2D.FromAngle(RotationRadians, Center);
        PointD localPx = pose.ApplyInverse(worldPt);
        double halfW = (Width * 0.5) + extraMarginPx;
        double halfH = (Height * 0.5) + extraMarginPx;

        if (Math.Abs(localPx.X) > halfW || Math.Abs(localPx.Y) > halfH)
        {
            return false;
        }

        if (Shape == RoiShapeKind.Rectangle) return true;

        if (Shape == RoiShapeKind.Circle)
        {
            double nx = localPx.X / Math.Max(1e-6, halfW);
            double ny = localPx.Y / Math.Max(1e-6, halfH);
            return (nx * nx) + (ny * ny) <= 1.0;
        }

        PointD uv = new(localPx.X / Math.Max(1e-6, Width), localPx.Y / Math.Max(1e-6, Height));
        return ContainsInPolygon(NormalizedVertices, uv);
    }

    private static bool ContainsInPolygon(ImmutableArray<PointD> poly, PointD pt)
    {
        if (poly.IsDefaultOrEmpty || poly.Length < 3) return false;
        bool inside = false;
        for (int i = 0, j = poly.Length - 1; i < poly.Length; j = i++)
        {
            PointD a = poly[i];
            PointD b = poly[j];
            bool crossesY = (a.Y > pt.Y) != (b.Y > pt.Y);
            if (crossesY)
            {
                double intersectX = ((b.X - a.X) * (pt.Y - a.Y) / (b.Y - a.Y)) + a.X;
                if (pt.X < intersectX) inside = !inside;
            }
        }
        return inside;
    }

    public double Support(PointD worldDir)
    {
        double cos = Math.Cos(RotationRadians);
        double sin = Math.Sin(RotationRadians);
        double lx = (worldDir.X * cos) + (worldDir.Y * sin);
        double ly = (-worldDir.X * sin) + (worldDir.Y * cos);
        double halfW = Width * 0.5;
        double halfH = Height * 0.5;

        if (Shape == RoiShapeKind.Circle)
        {
            return Math.Sqrt((halfW * halfW * lx * lx) + (halfH * halfH * ly * ly));
        }

        if (Shape == RoiShapeKind.Polygon && !NormalizedVertices.IsDefaultOrEmpty && NormalizedVertices.Length >= 3)
        {
            double maxProj = 0.0;
            foreach (PointD uv in NormalizedVertices)
            {
                double proj = (uv.X * Width * lx) + (uv.Y * Height * ly);
                if (proj > maxProj) maxProj = proj;
            }
            return maxProj;
        }

        return (halfW * Math.Abs(lx)) + (halfH * Math.Abs(ly));
    }

    public BoundsD WorldBounds
    {
        get
        {
            double ex = Support(new PointD(1.0, 0.0));
            double ey = Support(new PointD(0.0, 1.0));
            return new BoundsD(Center.X - ex, Center.Y - ey, Center.X + ex, Center.Y + ey);
        }
    }

    public PointD[] GetWorldCorners() =>
    [
        ToWorld(-0.5, -0.5),
        ToWorld(+0.5, -0.5),
        ToWorld(+0.5, +0.5),
        ToWorld(-0.5, +0.5)
    ];

    public PointD[] GetWorldSamplePoints()
    {
        if (Shape == RoiShapeKind.Polygon && !NormalizedVertices.IsDefaultOrEmpty && NormalizedVertices.Length >= 3)
        {
            var pts = new PointD[NormalizedVertices.Length];
            for (int i = 0; i < pts.Length; i++)
            {
                pts[i] = ToWorld(NormalizedVertices[i].X, NormalizedVertices[i].Y);
            }
            return pts;
        }

        if (Shape == RoiShapeKind.Circle)
        {
            var pts = new PointD[16];
            for (int i = 0; i < 16; i++)
            {
                double ang = (Math.PI * 2.0 * i) / 16.0;
                pts[i] = ToWorld(0.5 * Math.Cos(ang), 0.5 * Math.Sin(ang));
            }
            return pts;
        }

        return GetWorldCorners();
    }

    public RoiGeometry Translate(PointD delta) => this with { Center = Center + delta };
    public RoiGeometry WithRotation(double newRadians) => this with { RotationRadians = newRadians };
}

public static class PolygonClosure
{
    public static bool SegmentsIntersect(PointD a, PointD b, PointD c, PointD d)
    {
        PointD r = b - a;
        PointD s = d - c;
        double denom = PointD.Cross(r, s);
        if (Math.Abs(denom) < 1e-9) return false;

        double t = PointD.Cross(c - a, s) / denom;
        double u = PointD.Cross(c - a, r) / denom;
        return t > 1e-5 && t < 1.0 - 1e-5 && u > 1e-5 && u < 1.0 - 1e-5;
    }

    public static bool CanAddPoint(IReadOnlyList<PointD> existing, PointD candidate)
    {
        if (existing.Count < 2) return true;
        PointD last = existing[^1];
        for (int i = 0; i < existing.Count - 2; i++)
        {
            if (SegmentsIntersect(existing[i], existing[i + 1], last, candidate))
                return false;
        }
        return true;
    }

    public static bool TryCreateClosedPolygon(IReadOnlyList<PointD> worldPts, out RoiGeometry geometry)
    {
        geometry = RoiGeometry.CreateRectangle(new PointD(0, 0), 10, 10);
        if (worldPts.Count < 3) return false;

        for (int i = 1; i < worldPts.Count - 2; i++)
        {
            if (SegmentsIntersect(worldPts[i], worldPts[i + 1], worldPts[^1], worldPts[0]))
                return false;
        }

        double minX = double.MaxValue, minY = double.MaxValue;
        double maxX = double.MinValue, maxY = double.MinValue;
        foreach (PointD p in worldPts)
        {
            if (p.X < minX) minX = p.X;
            if (p.Y < minY) minY = p.Y;
            if (p.X > maxX) maxX = p.X;
            if (p.Y > maxY) maxY = p.Y;
        }

        double w = Math.Max(8.0, maxX - minX);
        double h = Math.Max(8.0, maxY - minY);
        PointD center = new((minX + maxX) * 0.5, (minY + maxY) * 0.5);

        var normBuilder = ImmutableArray.CreateBuilder<PointD>(worldPts.Count);
        foreach (PointD p in worldPts)
        {
            normBuilder.Add(new PointD((p.X - center.X) / w, (p.Y - center.Y) / h));
        }

        geometry = new RoiGeometry(RoiShapeKind.Polygon, center, w, h, 0.0, normBuilder.MoveToImmutable());
        return true;
    }
}

Step 3 — Step3_RoiMotionBounds.cs ClampTranslation (40회 이분법 + 벽면 정사영) & ClampRotation 
using System;

namespace MiniVisionCanvas;

public static class RoiMotionBounds
{
    public static bool ContainsGeometry(RoiGeometry parent, RoiGeometry child)
    {
        foreach (PointD pt in child.GetWorldSamplePoints())
        {
            if (!parent.Contains(pt)) return false;
        }
        return true;
    }

    public static PointD ClampTranslation(RoiGeometry parent, RoiGeometry child, PointD rawDelta)
    {
        if (ContainsGeometry(parent, child.Translate(rawDelta)))
        {
            return rawDelta;
        }

        double t = FindMaxFreeTranslationFactor(parent, child, rawDelta);
        PointD safeDelta = rawDelta * t;
        RoiGeometry atWall = child.Translate(safeDelta);

        PointD remaining = rawDelta - safeDelta;
        PointD slideDelta = ProjectAlongParentWall(parent, atWall, remaining);
        return safeDelta + slideDelta;
    }

    private static double FindMaxFreeTranslationFactor(RoiGeometry parent, RoiGeometry child, PointD delta)
    {
        double low = 0.0;
        double high = 1.0;
        for (int iter = 0; iter < 40; iter++)
        {
            double mid = (low + high) * 0.5;
            if (ContainsGeometry(parent, child.Translate(delta * mid)))
                low = mid;
            else
                high = mid;
        }
        return low;
    }

    private static PointD ProjectAlongParentWall(RoiGeometry parent, RoiGeometry childAtWall, PointD remaining)
    {
        double cos = Math.Cos(parent.RotationRadians);
        double sin = Math.Sin(parent.RotationRadians);
        PointD axisU = new(cos, sin);
        PointD axisV = new(-sin, cos);

        PointD projU = axisU * PointD.Dot(remaining, axisU);
        double tU = FindMaxFreeTranslationFactor(parent, childAtWall, projU);
        PointD appliedU = projU * tU;

        RoiGeometry afterU = childAtWall.Translate(appliedU);
        PointD projV = axisV * PointD.Dot(remaining, axisV);
        double tV = FindMaxFreeTranslationFactor(parent, afterU, projV);
        PointD appliedV = projV * tV;

        return appliedU + appliedV;
    }

    public static double ClampRotation(RoiGeometry parent, RoiGeometry child, double rawDeltaRadians)
    {
        if (ContainsGeometry(parent, child.WithRotation(child.RotationRadians + rawDeltaRadians)))
        {
            return rawDeltaRadians;
        }

        double low = 0.0;
        double high = 1.0;
        for (int iter = 0; iter < 40; iter++)
        {
            double mid = (low + high) * 0.5;
            if (ContainsGeometry(parent, child.WithRotation(child.RotationRadians + (rawDeltaRadians * mid))))
                low = mid;
            else
                high = mid;
        }
        return rawDeltaRadians * low;
    }
}

using System;
using System.Collections.Frozen;
using System.Collections.Generic;
using System.Collections.Immutable;
using System.Numerics;

namespace MiniVisionCanvas;

public readonly record struct BvhNodePath
{
    private readonly uint _valueMinusOne;

    public BvhNodePath(uint value)
    {
        ArgumentOutOfRangeException.ThrowIfZero(value);
        _valueMinusOne = value - 1;
    }

    public static BvhNodePath Root => default;
    public uint Value => _valueMinusOne + 1;
    public int Depth => 31 - BitOperations.LeadingZeroCount(Value);

    public BvhNodePath LeftChild() => new(Value << 1);
    public BvhNodePath RightChild() => new((Value << 1) | 1);
    public bool StepIsRight(int level) => ((Value >> (Depth - 1 - level)) & 1) != 0;
}

public sealed class RoiIdSet
{
    public static RoiIdSet Empty { get; } = new(FrozenSet<Guid>.Empty, ImmutableHashSet<Guid>.Empty, ImmutableHashSet<Guid>.Empty, 0);

    private readonly FrozenSet<Guid> _baseIds;
    private readonly ImmutableHashSet<Guid> _added;
    private readonly ImmutableHashSet<Guid> _removed;

    private RoiIdSet(FrozenSet<Guid> baseIds, ImmutableHashSet<Guid> added, ImmutableHashSet<Guid> removed, int count)
    {
        _baseIds = baseIds;
        _added = added;
        _removed = removed;
        Count = count;
    }

    public int Count { get; }
    public bool IsEmpty => Count == 0;

    public bool Contains(Guid id) =>
        _added.Contains(id) || (_baseIds.Contains(id) && !_removed.Contains(id));

    public RoiIdSet Add(Guid id) =>

    private sealed record Node(BvhNodePath Path, BoundsD Bounds, Node? Left, Node? Right, Guid? LeafId);

    private readonly Node? _root;

    private RoiSpatialIndex(Node? root) => _root = root;

    public static RoiSpatialIndex Empty { get; } = new(null);

    public static RoiSpatialIndex Build(IReadOnlyList<Entry> entries)
    {
        if (entries.Count == 0) return Empty;
        var mutable = new List<Entry>(entries);
        return new RoiSpatialIndex(BuildNode(mutable, 0, mutable.Count, BvhNodePath.Root));
    }

    private static Node BuildNode(List<Entry> items, int start, int count, BvhNodePath path)
    {
        if (count == 1)
        {
            Entry leaf = items[start];
            return new Node(path, leaf.Bounds, null, null, leaf.Id);
        }

        BoundsD total = items[start].Bounds;
        for (int i = start + 1; i < start + count; i++)
        {
            total = total.Union(items[i].Bounds);
        }

        bool splitX = total.Width >= total.Height;
        items.Sort(start, count, Comparer<Entry>.Create((a, b) =>
            splitX ? a.Bounds.Center.X.CompareTo(b.Bounds.Center.X)
                   : a.Bounds.Center.Y.CompareTo(b.Bounds.Center.Y)));

        int mid = count / 2;
        Node left = BuildNode(items, start, mid, path.LeftChild());
        Node right = BuildNode(items, start + mid, count - mid, path.RightChild());
        return new Node(path, total, left, right, null);
    }

    public List<Guid> QueryPoint(PointD pt)
    {
        var results = new List<Guid>();
        QueryPointRecursive(_root, pt, results);
        return results;
    }

    private static void QueryPointRecursive(Node? node, PointD pt, List<Guid> results)
    {
        if (node is null || !node.Bounds.Contains(pt)) return;
        if (node.LeafId is { } id)
        {
            results.Add(id);
            return;
        }
        QueryPointRecursive(node.Left, pt, results);
        QueryPointRecursive(node.Right, pt, results);
    }
}

public readonly record struct RepeatedRoiGrid(
    PointD Origin,
    double CellWidth,
    double CellHeight,
    double StepX,
    double StepY,
    int Columns,
    int Rows)
{
    public long TotalCellCount => (long)Columns * Rows;

    public BoundsD TotalBounds => new(
        Origin.X,
        Origin.Y,
        Origin.X + ((Columns - 1) * StepX) + CellWidth,
        Origin.Y + ((Rows - 1) * StepY) + CellHeight);

    public BoundsD GetCellBounds(int row, int col)
    {
        double left = Origin.X + (col * StepX);
        double top = Origin.Y + (row * StepY);
        return new BoundsD(left, top, left + CellWidth, top + CellHeight);
    }

    public bool TryHitCell(PointD worldPt, out int row, out int col, out long cellIndex)
    {
        row = 0;
        col = 0;
        cellIndex = -1;
        if (Columns <= 0 || Rows <= 0) return false;

        col = (int)Math.Floor((worldPt.X - Origin.X) / StepX);
        row = (int)Math.Floor((worldPt.Y - Origin.Y) / StepY);
        if (col < 0 || col >= Columns || row < 0 || row >= Rows) return false;

        BoundsD cellBounds = GetCellBounds(row, col);
        if (!cellBounds.Contains(worldPt)) return false;

        cellIndex = ((long)row * Columns) + col;
        return true;
    }

    public (int r0, int r1, int c0, int c1) GetVisibleRange(BoundsD visibleWorld)
    {
        int c0 = Math.Clamp((int)Math.Floor((visibleWorld.Left - Origin.X - CellWidth) / StepX), 0, Columns - 1);
        int c1 = Math.Clamp((int)Math.Ceiling((visibleWorld.Right - Origin.X) / StepX), 0, Columns - 1);
        int r0 = Math.Clamp((int)Math.Floor((visibleWorld.Top - Origin.Y - CellHeight) / StepY), 0, Rows - 1);
        int r1 = Math.Clamp((int)Math.Ceiling((visibleWorld.Bottom - Origin.Y) / StepY), 0, Rows - 1);
        return (r0, r1, c0, c1);
    }
}

public sealed record VirtualGridSlice(
    Guid Id,
    string Name,
    RepeatedRoiGrid Grid,
    ImmutableDictionary<long, CellOverride> CellOverrides,
    ImmutableHashSet<long> DeletedCells)
{
    public long ActiveCellCount => Grid.TotalCellCount - DeletedCells.Count;

    public RoiGeometry GetCellGeometry(long cellIndex)
    {
        int row = (int)(cellIndex / Grid.Columns);
        int col = (int)(cellIndex % Grid.Columns);
        BoundsD baseBounds = Grid.GetCellBounds(row, col);
        PointD center = baseBounds.Center;
        double w = Grid.CellWidth;
        double h = Grid.CellHeight;
        double rot = 0.0;

        if (CellOverrides.TryGetValue(cellIndex, out CellOverride ov))
        {
            center += ov.Offset;
            w = Math.Max(2.0, w + ov.WidthDelta);
            h = Math.Max(2.0, h + ov.HeightDelta);
            rot = ov.RotationRadians;
        }
        return RoiGeometry.CreateRectangle(center, w, h, rot);
    }

    public bool TryHit(PointD worldPt, out long hitCellIndex)
    {
        foreach (var kv in CellOverrides)
        {
            if (!DeletedCells.Contains(kv.Key) && GetCellGeometry(kv.Key).Contains(worldPt))
            {
                hitCellIndex = kv.Key;
                return true;
            }
        }

        if (Grid.TryHitCell(worldPt, out _, out _, out long idx)
            && !DeletedCells.Contains(idx)
            && !CellOverrides.ContainsKey(idx))
        {
            hitCellIndex = idx;
            return true;
        }

        hitCellIndex = -1;
        return false;
    }
}

            {
                Version = session.Document.Version + 1,
                RepeatedBatches = session.Document.RepeatedBatches.SetItem(idx, updated)

        }

        if (session.Document.SelectedIds.IsEmpty) return false;

        foreach (Guid id in moving)
        {

            }
        }
        session.CommitDocument(session.Document.WithUpdatedRois(builder.ToImmutable()));
        return true;
    }

    {
        _activeGesture.Cancel();
        _activeGesture = NoGesture.Instance;
        _polygonDraft.Clear();
        session.SetDraftPolygon([]);
    }

public static class TilePlanner
{
    public const int TileSize = 512;

    public static int SelectPyramidLevel(double scale, int maxLevel = 6)
    {
        if (scale >= 1.0) return 0;
        int level = (int)Math.Floor(Math.Log2(1.0 / scale));
        return Math.Clamp(level, 0, maxLevel);
    }

    public static void AverageRowSimd(ReadOnlySpan<int> rowA, ReadOnlySpan<int> rowB, Span<int> dst)
    {
        int i = 0;
        Vector256<int> one = Vector256.Create(1);
        for (; i + 8 <= dst.Length; i += 8)
        {
            Vector256<int> va = Vector256.Create(rowA.Slice(i, 8));
            Vector256<int> vb = Vector256.Create(rowB.Slice(i, 8));
            Vector256<int> avg = (va + vb + one) >> 1;
            avg.CopyTo(dst.Slice(i, 8));
        }
        for (; i < dst.Length; i++)
        {
            dst[i] = (rowA[i] + rowB[i] + 1) >> 1;
        }
    }
}

                dc.DrawLine(pen, new Point(a.X, a.Y), new Point(b.X, b.Y));
            }
            foreach (PointD p in poly)
            {
                PointD s = vp.ImageToScreen(p);
                dc.DrawEllipse(Brushes.Lime, null, new Point(s.X, s.Y), 4, 4);
            }
        }
    }

    private void DrawMinimap(DrawingContext dc, DocumentState doc, ViewportTransform vp)
    {
        if (ActualWidth < 240 || ActualHeight < 180) return;
        Rect mini = new(ActualWidth - 156, ActualHeight - 116, 140, 100);
        dc.DrawRectangle(new SolidColorBrush(Color.FromArgb(210, 15, 23, 42)), new Pen(Brushes.SlateGray, 1.0), mini);

        double sx = mini.Width / Math.Max(1.0, doc.ImageWidth);
        double sy = mini.Height / Math.Max(1.0, doc.ImageHeight);
        BoundsD vis = vp.ScreenToImage(new BoundsD(0, 0, ActualWidth, ActualHeight));
        double vx = Math.Clamp(mini.Left + (vis.Left * sx), mini.Left, mini.Right);
        double vy = Math.Clamp(mini.Top + (vis.Top * sy), mini.Top, mini.Bottom);
        double vw = Math.Clamp(vis.Width * sx, 4.0, mini.Right - vx);
        double vh = Math.Clamp(vis.Height * sy, 4.0, mini.Bottom - vy);
        dc.DrawRectangle(null, new Pen(Brushes.Cyan, 1.5), new Rect(vx, vy, vw, vh));
    }
}

    private void DeleteSelection()
    {
        if (Controller.NavigationOnly) return;
        if (Session.Document.SelectedRepeatedCell is { } selCell)
        {
            int idx = Session.Document.FindBatchIndex(selCell.BatchId);
            if (idx >= 0)
            {
                VirtualGridSlice b = Session.Document.RepeatedBatches[idx];
                VirtualGridSlice nextB = b with { DeletedCells = b.DeletedCells.Add(selCell.CellIndex) };
                Session.CommitDocument(Session.Document with
                {
                    Version = Session.Document.Version + 1,
                    RepeatedBatches = Session.Document.RepeatedBatches.SetItem(idx, nextB),
                    SelectedRepeatedCell = null
                });
                Status = $"1억 배치 중 #{selCell.CellIndex:N0}번 셀 개별 삭제 완료 (남은 총 ROI: {Session.Document.TotalLogicalRoiCount:N0}개)";
            }
            return;
        }

        if (Session.Document.SelectedIds.IsEmpty) return;
        ImmutableHashSet<Guid> toDelete = Session.Document.CollectSubtreeIds(Session.Document.SelectedIds);
        var builder = Session.Document.Rois.ToBuilder();
        foreach (Guid id in toDelete) builder.Remove(id);
        Session.CommitDocument(Session.Document.WithUpdatedRois(builder.ToImmutable(), ImmutableHashSet<Guid>.Empty));
        Status = $"{toDelete.Count}개 ROI 삭제 완료";
    }

    public bool HandleKey(Key key, ModifierKeys modifiers)
    {
        if (key == Key.Escape)
        {
            Controller.Cancel();
            Status = "작업 취소됨 (Escape)";
            return true;
        }
        if (Controller.NavigationOnly) return false;

        if (key == Key.Enter && Controller.Tool == EditorTool.DrawPolygon)
        {
            CompletePolygonCommand.Execute(null);
            return true;
        }

        double step = modifiers.HasFlag(ModifierKeys.Shift) ? 10.0 : 1.0;
        PointD delta = key switch
        {
            Key.Left  => new PointD(-step, 0.0),
            Key.Right => new PointD(+step, 0.0),
            Key.Up    => new PointD(0.0, -step),
            Key.Down  => new PointD(0.0, +step),
            _ => default
        };
        if ((delta.X != 0.0 || delta.Y != 0.0) && Controller.TryMoveSelectionBy(delta))
        {
            Status = $"방향키 미세 이동: ({delta.X:+0;-0;0}, {delta.Y:+0;-0;0}) px";
            return true;
        }

        bool ctrl = modifiers.HasFlag(ModifierKeys.Control);
        bool shift = modifiers.HasFlag(ModifierKeys.Shift);
        ICommand? cmd = (key, ctrl, shift) switch
        {
            (Key.Delete, _, _) => DeleteCommand,
            (Key.Z, true, false) => UndoCommand,
            (Key.Y, true, _) or (Key.Z, true, true) => RedoCommand,
            _ => null
        };
        if (cmd is null || !cmd.CanExecute(null)) return false;
        cmd.Execute(null);
        return true;
    }
